def append_epss_history(conn: sqlite3.Connection, cve_id: str, score: float) -> bool:
    """Record one EPSS data point per CVE per day. Returns True if inserted."""
    from datetime import datetime, timezone

    today = datetime.now(timezone.utc).strftime("%Y-%m-%d")
    cur = conn.execute(
        "INSERT OR IGNORE INTO epss_history (cve_id, score, recorded_at)"
        " VALUES (?, ?, ?)",
        (cve_id, score, today),
    )
    return cur.rowcount > 0
```

Note: `recorded_at` stores the date only so the `UNIQUE (cve_id, recorded_at)` constraint enforces one row per day via `INSERT OR IGNORE`.

    """Backfill EPSS scores + append history for all known CVEs."""
    all_cves = conn.execute("SELECT id, epss_score FROM cve").fetchall()
    updated = 0
    for cve_id, current in all_cves:

    return updated

# horus/core/forecast.py
"""Exploitability forecasting — heuristic, ML-ready.

v0.11 ships a transparent heuristic. The epss_history + exploitation_outcome
tables accrue the data a v0.12 ML model will train on, behind this same
interface.
"""

from __future__ import annotations

from datetime import date

_DATE_FMT = "%Y-%m-%d"


def epss_velocity(cve_id: str, conn) -> float:
    """Δscore per day over the two most recent distinct history points.

    Returns 0.0 when fewer than two points exist.
    """
    rows = conn.execute(
        "SELECT score, recorded_at FROM epss_history"
        " WHERE cve_id = ? ORDER BY recorded_at DESC LIMIT 2",
        (cve_id,),
    ).fetchall()
    if len(rows) < 2:
        return 0.0
    (s_new, d_new), (s_old, d_old) = rows[0], rows[1]
    try:
        days = (
            date.fromisoformat(d_new) - date.fromisoformat(d_old)
        ).days
    except ValueError:
        return 0.0
    if days <= 0:
        return 0.0
    return (s_new - s_old) / days
```

```python
from .model import CVE

# Products whose ubiquity raises imminence (mirrors reputation's ubiquity set).
_UBIQUITY = (
    "nginx", "php", "wordpress", "openssl", "kubernetes", "openssh",
    "apache", "mysql", "chrome", "linux",
)


def compute_imminence(cve: CVE, *, epss_velocity: float) -> tuple[float, str]:
    """Heuristic 0–10 imminence score + coarse bucket.

    Weights are intentionally transparent; v0.12 replaces this body with a
    trained model behind the same signature.
    """
    epss = cve.epss_score or 0.0
    # Normalize velocity: 0.02/day (≈ 0→0.6 over a month) saturates to 1.0.
    velocity_norm = max(0.0, min(epss_velocity / 0.02, 1.0))
    poc_signal = min((cve.poc_source_count or 0) * 0.75, 1.5)
    cvss = (cve.cvss_score or 0.0) / 10.0
    desc = (cve.description or "").lower()
    ubiquity = 0.5 if any(p in desc for p in _UBIQUITY) else 0.0

    score = (
        epss * 3.0
        + velocity_norm * 2.0
        + (2.0 if cve.kev else 0.0)
        + poc_signal
        + cvss * 1.0
        + ubiquity
    )
    score = round(min(score, 10.0), 3)

    if score >= 7.5:
        bucket = "imminent"
    elif score >= 5.0:
        bucket = "weeks"
    elif score >= 2.5:
        bucket = "months"
    else:
        bucket = "unlikely"
    return score, bucket
```

# horus/enrichers/exposure.py
"""Enricher — real-world exposure (exposed host counts).

Optional and pluggable. Ships a Shodan adapter; no-ops cleanly when
SHODAN_API_KEY is unset (mirrors X-auth-absent behavior). Censys and others
can be added as more adapters behind ExposureAdapter.
"""

from __future__ import annotations

import json
import os
import sys
import urllib.parse
import urllib.request
from typing import Protocol

from ..config import EXPOSURE_MAX_QUERIES_PER_RUN
from ..core.model import CVE
from ..storage import db

NAME = "Exposure (Shodan)"
DEFAULT_ENABLED = True


class ExposureAdapter(Protocol):
    name: str

    def host_count(self, cve: CVE) -> int | None: ...


class ShodanAdapter:
    name = "shodan"

    def __init__(self, api_key: str) -> None:
        self._key = api_key

    def host_count(self, cve: CVE) -> int | None:
        # Shodan count API: search for the CVE id across the index.
        q = urllib.parse.quote(f"vuln:{cve.id}")
        url = (
            "https://api.shodan.io/shodan/host/count"
            f"?key={self._key}&query={q}"
        )
        try:
            with urllib.request.urlopen(url, timeout=30) as resp:
                data = json.loads(resp.read())
            return int(data.get("total", 0))
        except Exception:
            return None


def _build_adapter() -> ExposureAdapter | None:
    key = os.environ.get("SHODAN_API_KEY")
    if not key:
        return None
    return ShodanAdapter(key)


def enrich(ctx) -> None:
    adapter = _build_adapter()
    if adapter is None:
        print("  exposure: no SHODAN_API_KEY, skipping", file=sys.stderr)
        return

    # Prioritize the CVEs that matter: highest reputation first, capped.
    targets = sorted(
        ctx.cves, key=lambda c: c.reputation_score or 0.0, reverse=True
    )[:EXPOSURE_MAX_QUERIES_PER_RUN]

    count = 0
    with db.connect() as conn:
        for cve in targets:
            hosts = adapter.host_count(cve)
            if hosts is None:
                continue
            cve.exposed_host_count = hosts
            db.record_exposure(conn, cve.id, adapter.name, hosts)
            count += 1
    if count:
        print(f"  exposure: {count} CVEs scored via {adapter.name}", file=sys.stderr)
```

"""ML-ready outcome logging.

Accrues labeled history (published → first PoC → became KEV) so a v0.12 model

from __future__ import annotations

from datetime import datetime, timezone


def record_outcomes(conn) -> int:
    now = datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")
    rows = conn.execute(
        "SELECT c.id, c.published_at, c.kev, c.first_seen,"
        " (SELECT MIN(p.first_seen) FROM poc p"
        "  JOIN poc_cve pc ON pc.poc_url = p.url WHERE pc.cve_id = c.id)"
        " FROM cve c"
    ).fetchall()
    written = 0
    for cve_id, published_at, kev, first_seen, first_poc_at in rows:
        became_kev_at = first_seen if kev else None
        conn.execute(
            "INSERT INTO exploitation_outcome"
            " (cve_id, published_at, first_poc_at, became_kev_at,"
            "  first_exploited_signal_at, recorded_at)"
            " VALUES (?, ?, ?, ?, ?, ?)"
            " ON CONFLICT(cve_id) DO UPDATE SET"
            "  published_at=excluded.published_at,"
            "  first_poc_at=excluded.first_poc_at,"
            "  became_kev_at=COALESCE(exploitation_outcome.became_kev_at, excluded.became_kev_at),"
            "  recorded_at=excluded.recorded_at",
            (cve_id, published_at, first_poc_at, became_kev_at, None, now),
        )
        written += 1
    return written
```

Note `became_kev_at` uses `COALESCE(existing, new)` so the *first* observed KEV date is preserved across runs.
